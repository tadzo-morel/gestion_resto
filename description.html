        -String villePers
        -String loginPers
        -String passwordPers

        -Integer numCarteFidelite
        -Integer pointsFidelite
        -Date dateInscription
        +ajouterPoints(int) void

        +getReductions() double
    }

        -String droit

        +consulterStatistiques() void

    %% GESTION DES COMMANDES
    class Commande {
        -Long idCommande
        -Date dateCom
        -Integer heureCom
        -StatutCommande etatCommande
        -TypeCommande typeCommande
        -Double montantTotal
        +calculerMontantTotal() Double
        +ajouterLigne() void
    }
    
    class StatutCommande {
        <>
        EN_ATTENTE
        EN_PREPARATION
        PRETE
        EN_LIVRAISON
        LIVREE
        ANNULEE
    }
    
    class TypeCommande {
        <>
        SUR_PLACE
        A_EMPORTER
        LIVRAISON
    }
    
    class LigneCommande {
        -Long idLigne
        -Integer quantite
        -Double prixUnitaire
        -Double sousTotal
        +calculerSousTotal() Double
    }
    
    class Livraison {
        -Long idLiv
        -String adresseLiv
        -Date dateLiv
        -Integer heureLiv
        -StatutLivraison statut
        -Double fraisLivraison
        +assignerLivreur() void
    }
    
    class StatutLivraison {
        <>
        EN_ATTENTE
        ASSIGNEE
        EN_COURS
        LIVREE
    }
    
    class Facture {
        -Long idFacture
        -String numFacture
        -Date dateFacture
        -Double montantHT
        -Double montantTVA
        -Double montantTTC
        +genererFacture() void
    }
    
    class Paiement {
        -Long idPaiement
        -Date datePaiement
        -Double montant
        -ModePaiement mode
        +effectuerPaiement() Boolean
    }
    
    class ModePaiement {
        <>
        ESPECES
        CARTE_BANCAIRE
        MOBILE_MONEY
    }
